import { describe, expect, it } from "vitest";
import {
  checkoutSummary,
  createCheckoutGame,
  endCheckoutRun,
  endCheckoutTurn,
  isCheckable,
  nextTarget,
  restoreCheckoutGame,
  throwCheckoutDart,
  undoCheckoutDart,
  undoCheckoutTurn,
  type CheckoutConfig,
  type CheckoutGame,
} from "./checkout-engine";
import type { Dart } from "./engine";

const S = (value: number): Dart => ({ value, multiplier: 1 });
const D = (value: number): Dart => ({ value, multiplier: 2 });
const T = (value: number): Dart => ({ value, multiplier: 3 });
const miss = S(0);

/** Default: start 40 (a single double 20), 3 darts per attempt, 10 rounds. */
function config(overrides: Partial<CheckoutConfig> = {}, players = 1): CheckoutConfig {
  return {
    startTarget: 40,
    dartsPerAttempt: 3,
    rounds: 10,
    players: Array.from({ length: players }, (_, i) => ({
      id: `p${i + 1}`,
      name: `Player ${i + 1}`,
    })),
    ...overrides,
  };
}
const game = (overrides: Partial<CheckoutConfig> = {}, players = 1) =>
  createCheckoutGame(config(overrides, players));
function throwAll(g: CheckoutGame, darts: Dart[]): CheckoutGame {
  return darts.reduce((acc, d) => throwCheckoutDart(acc, d), g);
}

describe("checkable targets", () => {
  it("knows exactly which totals from 2 to 170 cannot be checked out", () => {
    const impossible = Array.from({ length: 169 }, (_, i) => i + 2).filter(
      (n) => !isCheckable(n),
    );
    expect(impossible).toEqual([159, 162, 163, 165, 166, 168, 169]);
    expect(isCheckable(1)).toBe(false);
    expect(isCheckable(171)).toBe(false);
    expect(isCheckable(40.5)).toBe(false);
  });
  it("raises the target by one and never skips", () => {
    expect(nextTarget(121)).toBe(122);
    expect(nextTarget(158)).toBe(159);
  });
});

describe("configuration", () => {
  it("accepts a normal setup", () => {
    expect(() => createCheckoutGame(config({ startTarget: 121 }, 4))).not.toThrow();
    expect(() => createCheckoutGame(config({ rounds: "infinite" }))).not.toThrow();
    expect(() => createCheckoutGame(config({ dartsPerAttempt: 15, rounds: 99 }))).not.toThrow();
  });
  it("rejects bad configurations", () => {
    expect(() => createCheckoutGame(config({ startTarget: 159 }))).toThrow(RangeError);
    expect(() => createCheckoutGame(config({ startTarget: 171 }))).toThrow(RangeError);
    expect(() => createCheckoutGame(config({ startTarget: 1 }))).toThrow(RangeError);
    expect(() => createCheckoutGame(config({ dartsPerAttempt: 4 }))).toThrow(RangeError);
    expect(() => createCheckoutGame(config({ rounds: 0 }))).toThrow(RangeError);
    expect(() => createCheckoutGame(config({ rounds: 100 }))).toThrow(RangeError);
    expect(() => createCheckoutGame(config({ rounds: 2.5 }))).toThrow(RangeError);
    expect(() => createCheckoutGame(config({ rounds: "forever" as never }))).toThrow(RangeError);
    expect(() => createCheckoutGame(config({}, 0))).toThrow(RangeError);
    expect(() => createCheckoutGame(config({}, 5))).toThrow(RangeError);
    const dup = config({}, 2);
    dup.players[1].id = dup.players[0].id;
    expect(() => createCheckoutGame(dup)).toThrow(RangeError);
  });
});

  it("ends the attempt on the checkout dart and raises the target", () => {
    const g = throwCheckoutDart(game(), D(20));
    expect(g.attempts).toEqual([
      { number: 1, target: 40, darts: 1, outcome: "cleared", by: 0 },

    expect(g.target).toBe(41);
    expect(g.remaining).toBe(41);
    expect(g.attempt).toBe(2);
    expect(g.dartsUsed).toBe(0);
    expect(g.status).toBe("playing");
    expect(g.notice).toBe("Cleared 40 in 1 dart.");
  });
  it("counts the bullseye as a double", () => {
    const g = throwCheckoutDart(game({ startTarget: 50 }), D(25));
    expect(g.best).toBe(50);
  });
  it("clears over several darts and turns", () => {
    const g = throwAll(game({ dartsPerAttempt: 6, startTarget: 100 }), [
      T(20), // 40 left
      miss,
      miss,
      S(20), // 20 left
      D(10),

    expect(g.attempts[0]).toMatchObject({ target: 100, darts: 5, outcome: "cleared" });

describe("double out and bust", () => {
  it("is a bust to finish on a single, and the remaining score goes back", () => {
    const g = throwAll(game({ dartsPerAttempt: 6 }), [S(20), S(20)]);
    expect(g.remaining).toBe(40);
    expect(g.turns[0]).toMatchObject({ bust: true, unthrown: 1, complete: true });
    // The bust burned the unthrown dart: 3 of 6 used, attempt still open.
    expect(g.dartsUsed).toBe(3);
    expect(g.attempt).toBe(1);
    expect(g.turnDarts).toEqual([]);
    const cleared = throwCheckoutDart(g, D(20));
    expect(cleared.attempts[0]).toMatchObject({ darts: 4, outcome: "cleared" });
  });
  it("is a bust to land on 1 or go below zero", () => {
    const on1 = throwAll(game({ dartsPerAttempt: 6 }), [S(19), S(20)]);
    expect(on1.remaining).toBe(40);
    expect(on1.turns[0].bust).toBe(true);
    const below = throwCheckoutDart(game({ dartsPerAttempt: 6 }), T(20));
    expect(below.remaining).toBe(40);
    expect(below.turns[0].bust).toBe(true);
  });
  it("restores the score from the start of the turn, not the attempt", () => {
    const g = throwAll(game({ dartsPerAttempt: 9, startTarget: 100 }), [
      miss,
      miss,
      miss, // turn 1, still 100
      T(20), // 40 left
      S(20), // 20 left
      S(20), // zero on a single: bust, back to 100 (turn start)
    ]);
    expect(g.remaining).toBe(100);
  });
  it("fails the attempt when a bust burns the last darts", () => {
    const g = throwAll(game({ dartsPerAttempt: 3 }), [S(20), S(20)]);
    expect(g.attempts[0]).toMatchObject({ outcome: "missed", darts: 3, by: null });
    expect(g.attempt).toBe(2);
  });
});

describe("missing a target and the safehouse", () => {
  it("falls back to the start target when nothing was banked", () => {
    const g = throwAll(game(), [miss, miss, miss]);
    expect(g.attempts[0].outcome).toBe("missed");
    expect(g.target).toBe(40);
    expect(g.notice).toBe("Missed 40. Back to 40.");
  });
  it("moves the safehouse up for a checkout in three darts or fewer", () => {
    let g = throwCheckoutDart(game(), D(20)); // 40 in 1 dart
    g = throwAll(g, [S(1), D(20)]); // 41 in 2 darts
    expect(g.best).toBe(41);
    expect(g.safehouse).toBe(41);
    expect(g.target).toBe(42);
    g = throwAll(g, [miss, miss, miss]);
    expect(g.target).toBe(41);
  });
  it("does not move the safehouse for a checkout that took more than three darts", () => {
    let g = game({ dartsPerAttempt: 6 });
    g = throwCheckoutDart(g, D(20)); // 40 in 1 dart, safehouse 40
    g = throwAll(g, [miss, miss, miss, S(1), D(20)]); // 41 in 5 darts
    expect(g.best).toBe(41);
    expect(g.safehouse).toBe(40);
    expect(g.target).toBe(42);
    g = throwAll(g, [miss, miss, miss, miss, miss, miss]);
    expect(g.attempts.at(-1)).toMatchObject({ target: 42, outcome: "missed" });
    expect(g.target).toBe(40);
  });
  it("cannot clear 159 in three darts, but can in six", () => {
    const at158 = throwAll(game({ startTarget: 158 }), [T(20), T(20), D(19)]);
    expect(at158.target).toBe(159);
    const failed = throwAll(at158, [T(20), T(20), D(19)]);
    expect(failed.attempts.at(-1)).toMatchObject({ target: 159, outcome: "missed" });
    expect(failed.target).toBe(158);

    const six = throwAll(game({ startTarget: 158, dartsPerAttempt: 6 }), [T(20), T(20), D(19)]);
    const cleared = throwAll(six, [T(20), T(20), S(19), D(10)]);
    expect(cleared.attempts.at(-1)).toMatchObject({ target: 159, darts: 4, outcome: "cleared" });
    expect(cleared.target).toBe(160);
  });
});

import { checkoutSuggestions, validateDart, type Dart } from "./engine";

export type CheckoutRounds = number | "infinite";
export type CheckoutConfig = {
  /** The first target, 2 to 170, checkable with three darts and a double. */
  startTarget: number;
  /** Darts for one attempt: 3, 6, 9, 12 or 15. */
  dartsPerAttempt: number;
  /** A whole number of attempts from 1 to 99, or "infinite". */
  rounds: CheckoutRounds;
  players: { id: string; name: string }[];
};
export type CheckoutEvent =
  | { type: "dart"; dart: Dart }
  | { type: "end-turn" }
  | { type: "end-run" };
export type CheckoutAttempt = {
  number: number;
  target: number;
  /** Darts of the budget used, counting darts burned by a bust or an early end of turn. */
  darts: number;
  outcome: "cleared" | "missed";
  /** Index of the player who threw the checkout dart. */
  by: number | null;
};
export type CheckoutTurn = {
  playerIndex: number;
  attempt: number;
  darts: Dart[];
  /** Darts of the turn not thrown (bust or early end). They still burn budget. */
  unthrown: number;
  bust: boolean;
  complete: boolean;
};
export type CheckoutGame = {
  config: CheckoutConfig;
  players: { id: string; name: string }[];
  /** Target of the current attempt. */
  target: number;
  remaining: number;
  /** Remaining at the start of the current turn, restored by a bust. */
  turnStart: number;
  /** Fall-back target after a missed attempt. */
  safehouse: number;
  /** Highest target cleared so far, 0 when none. */
  best: number;
  /** 1-based number of the current attempt. */
  attempt: number;
  dartsUsed: number;
  activePlayer: number;
  turnDarts: Dart[];
  turns: CheckoutTurn[];
  /** Finished attempts only, in order. */
  attempts: CheckoutAttempt[];
  status: "playing" | "complete";
  notice: string | null;
  events: CheckoutEvent[];
};

export const CHECKOUT_DARTS = [3, 6, 9, 12, 15];
export const MAX_TARGET = 170;

/** True when the score can be checked out with three darts ending on a double. */
export function isCheckable(score: number): boolean {
  return (
    Number.isInteger(score) &&
    score >= 2 &&
    score <= MAX_TARGET &&
    checkoutSuggestions(score, 3, true).length > 0
  );
}

/** The target rule lives here: a success raises the target by 1, never skipping. */
export function nextTarget(cleared: number): number {
  return cleared + 1;
}

export function createCheckoutGame(config: CheckoutConfig): CheckoutGame {
  if (
    !isCheckable(config.startTarget) ||
    !CHECKOUT_DARTS.includes(config.dartsPerAttempt) ||
    !(
      config.rounds === "infinite" ||
      (Number.isInteger(config.rounds) &&
        (config.rounds as number) >= 1 &&
        (config.rounds as number) <= 99)
    ) ||
    !Array.isArray(config.players) ||
    config.players.length < 1 ||
    config.players.length > 4 ||
    config.players.some(
      (p) =>
        typeof p.id !== "string" ||
        !p.id ||
        typeof p.name !== "string" ||
        !p.name.trim() ||
        p.name.length > 40,
    ) ||
    new Set(config.players.map((p) => p.id)).size !== config.players.length
  )
    throw new RangeError("Invalid 121 Checkout configuration");
  const copy: CheckoutConfig = {
    ...config,
    players: config.players.map((p) => ({ ...p })),
  };
  return {
    config: copy,
    players: copy.players.map((p) => ({ ...p })),
    target: copy.startTarget,
    remaining: copy.startTarget,
    turnStart: copy.startTarget,
    safehouse: copy.startTarget,
    best: 0,
    attempt: 1,
    dartsUsed: 0,
    activePlayer: 0,
    turnDarts: [],
    turns: [],
    attempts: [],
    status: "playing",
    notice: null,
    events: [],
  };
}

const plural = (n: number, word: string) => `${n} ${word}${n === 1 ? "" : "s"}`;

/** Closes the attempt, applies the target and safehouse rules and opens the next one. */
function endAttempt(
  game: CheckoutGame,
  outcome: "cleared" | "missed",
  by: number | null,
) {
  const turn = game.turns[game.turns.length - 1];
  if (turn) turn.complete = true;
  const target = game.target;
  game.attempts.push({
    number: game.attempt,
    target,
    darts: game.dartsUsed,
    outcome,
    by,
  });
  if (outcome === "cleared") {
    game.best = Math.max(game.best, target);
    if (game.dartsUsed <= 3) game.safehouse = target;
    game.notice = `Cleared ${target} in ${plural(game.dartsUsed, "dart")}.`;
    if (target >= MAX_TARGET) {
      game.status = "complete";
      game.turnDarts = [];
      return;
    }
    game.target = nextTarget(target);
  } else {
    game.target = game.safehouse;
    game.notice = `Missed ${target}. Back to ${game.safehouse}.`;
  }
  if (
    game.config.rounds !== "infinite" &&
    game.attempts.length >= game.config.rounds
  ) {
    game.status = "complete";
    game.turnDarts = [];
    return;
  }
  game.attempt += 1;
  game.remaining = game.target;
  game.turnStart = game.target;
  game.dartsUsed = 0;
  game.turnDarts = [];
  game.activePlayer = (game.activePlayer + 1) % game.players.length;
}

/** The turn is over without a checkout: burn its unthrown darts, then move on. */
function endTurn(game: CheckoutGame) {
  const turn = game.turns[game.turns.length - 1];
  turn.unthrown = 3 - game.turnDarts.length;
  game.dartsUsed += turn.unthrown;
  turn.complete = true;
  if (game.dartsUsed >= game.config.dartsPerAttempt)
    endAttempt(game, "missed", null);
  else {
    game.activePlayer = (game.activePlayer + 1) % game.players.length;
    game.turnDarts = [];
  }
}

/** Replay owns all its objects; public transitions never mutate their input. */
function replay(config: CheckoutConfig, events: CheckoutEvent[]): CheckoutGame {
  const game = createCheckoutGame(config);
  for (const event of events) {
    if (game.status === "complete")
      throw new RangeError("Saved events continue after the run ended.");
    game.notice = null;
    if (event.type === "end-run") {
      game.events.push({ type: "end-run" });
      const turn = game.turns[game.turns.length - 1];
      if (turn) turn.complete = true;
      game.turnDarts = [];
      game.status = "complete";
      continue;
    }
    if (event.type === "end-turn") {
      if (!game.turnDarts.length) continue;
      game.events.push({ type: "end-turn" });
      endTurn(game);
      continue;
    }
    const dart = { ...event.dart };
    validateDart(dart);
    game.events.push({ type: "dart", dart });
    if (!game.turnDarts.length) {
      game.turnStart = game.remaining;
      game.turns.push({
        playerIndex: game.activePlayer,
        attempt: game.attempt,
        darts: [],
        unthrown: 0,
        bust: false,
        complete: false,
      });
    }
    const turn = game.turns[game.turns.length - 1];
    turn.darts.push(dart);
    game.turnDarts.push(dart);
    game.dartsUsed += 1;
    const next = game.remaining - dart.value * dart.multiplier;
    if (next === 0 && dart.multiplier === 2) {
      endAttempt(game, "cleared", game.activePlayer);
    } else if (next <= 1) {
      // Below zero, on 1, or on zero without a double: a bust.
      game.remaining = game.turnStart;
      turn.bust = true;
      endTurn(game);
    } else {
      game.remaining = next;
      if (game.turnDarts.length === 3) endTurn(game);
    }
  }
  return game;
}

export function throwCheckoutDart(game: CheckoutGame, dart: Dart): CheckoutGame {
  if (game.status === "complete") return game;
  validateDart(dart);
  return replay(game.config, [...game.events, { type: "dart", dart }]);
}

/** Finishing a turn early burns its unthrown darts from the attempt budget. */
export function endCheckoutTurn(game: CheckoutGame): CheckoutGame {
  if (game.status === "complete" || !game.turnDarts.length) return game;
  return replay(game.config, [...game.events, { type: "end-turn" }]);
}

/** Ends the run. A partly played attempt is not counted, its darts stay in the events. */
export function endCheckoutRun(game: CheckoutGame): CheckoutGame {
  if (game.status === "complete") return game;
  return replay(game.config, [...game.events, { type: "end-run" }]);
}

/** Removes an end of run first; otherwise the last dart and any end-turn after it. */
export function undoCheckoutDart(game: CheckoutGame): CheckoutGame {
  if (game.events[game.events.length - 1]?.type === "end-run")
    return replay(game.config, game.events.slice(0, -1));
  for (let i = game.events.length - 1; i >= 0; i--) {
    if (game.events[i].type === "dart")
      return replay(game.config, game.events.slice(0, i));
  }
  return game;
}

/** Removes an end of run, else the current partial turn, else the last completed turn. */
export function undoCheckoutTurn(game: CheckoutGame): CheckoutGame {
  if (game.events[game.events.length - 1]?.type === "end-run")
    return replay(game.config, game.events.slice(0, -1));
  const turn = game.turns[game.turns.length - 1];
  if (!turn) return game;
  const endMarker =
    game.events[game.events.length - 1]?.type === "end-turn" ? 1 : 0;
  return replay(
    game.config,
    game.events.slice(0, game.events.length - turn.darts.length - endMarker),
  );
}

/** Restore only legal events, rebuilding all derived state. */
export function restoreCheckoutGame(
  config: CheckoutConfig,
  events: CheckoutEvent[],
): CheckoutGame {
  if (!Array.isArray(events) || events.length > 50000)
    throw new RangeError("Invalid saved events");
  for (const event of events) {
    if (
      !event ||
      (event.type !== "dart" &&
        event.type !== "end-turn" &&
        event.type !== "end-run")
    )
      throw new RangeError("Invalid saved event");
    if (event.type === "dart") {
      if (!event.dart) throw new RangeError("Missing saved dart");
      validateDart(event.dart);
    }
  }
  return replay(config, events);
}

export function checkoutSummary(game: CheckoutGame) {
  return {
    best: game.best,
    attemptsPlayed: game.attempts.length,
    attemptsCleared: game.attempts.filter((a) => a.outcome === "cleared")
      .length,
    darts: game.turns.reduce((sum, t) => sum + t.darts.length, 0),
    attempts: game.attempts,
    players: game.players.map((p, i) => ({
      id: p.id,
      name: p.name,
      darts: game.turns
        .filter((t) => t.playerIndex === i)
        .reduce((sum, t) => sum + t.darts.length, 0),
      checkouts: game.attempts.filter((a) => a.by === i).length,
    })),
  };
}

import { isCheckable, type CheckoutRounds } from "./checkout-engine";

export type CheckoutSetupInput = {
  startMode: "121" | "custom";
  customStart: string;
  roundsMode: "10" | "custom" | "infinite";
  customRounds: string;
};
export type CheckoutSetupResult =
  | { ok: true; startTarget: number; rounds: CheckoutRounds }
  | { ok: false; error: string };

const whole = (text: string) => (/^\d{1,3}$/.test(text.trim()) ? Number(text) : NaN);

/** Turns the setup fields into a start target and a round count, or one plain error. */
export function parseCheckoutSetup(input: CheckoutSetupInput): CheckoutSetupResult {
  let startTarget = 121;
  if (input.startMode === "custom") {
    startTarget = whole(input.customStart);
    if (!isCheckable(startTarget))
      return {
        ok: false,
        error:
          "Start target must be 2 to 170 and possible to check out with three darts. 159, 162, 163, 165, 166, 168 and 169 are not.",
      };
  }
  let rounds: CheckoutRounds = 10;
  if (input.roundsMode === "infinite") rounds = "infinite";
  else if (input.roundsMode === "custom") {
    const n = whole(input.customRounds);
    if (!Number.isInteger(n) || n < 1 || n > 99)
      return { ok: false, error: "Rounds must be a whole number from 1 to 99." };
    rounds = n;
  }
  return { ok: true, startTarget, rounds };
}
